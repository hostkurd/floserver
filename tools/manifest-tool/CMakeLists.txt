add_library(floserver_manifest_build STATIC build.cpp publish.cpp signing.cpp)
target_include_directories(floserver_manifest_build PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated)
# templates.json goes into the tool, so a build never needs a file beside it.
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/templates.json FLOSERVER_TEMPLATES_JSON)
# MSVC refuses a string literal over about 16 KB (C2026), counted in bytes;
# split templates_json.h.in into pieces before it grows that far.
string(LENGTH "${FLOSERVER_TEMPLATES_JSON}" FLOSERVER_TEMPLATES_BYTES)
if(FLOSERVER_TEMPLATES_BYTES GREATER 15000)
    message(FATAL_ERROR "templates.json is ${FLOSERVER_TEMPLATES_BYTES} bytes; split the literal in templates_json.h.in")
endif()
string(FIND "${FLOSERVER_TEMPLATES_JSON}" ")FLOTPL" FLOSERVER_TEMPLATES_DELIMITER)
if(NOT FLOSERVER_TEMPLATES_DELIMITER EQUAL -1)
    message(FATAL_ERROR "templates.json holds the raw string delimiter )FLOTPL")
endif()
if(FLOSERVER_TEMPLATES_JSON MATCHES "@[A-Za-z_]+@")
    message(FATAL_ERROR "templates.json holds an @WORD@ that configure_file would replace")
endif()
configure_file(templates_json.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/templates_json.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/templates.json)
target_link_libraries(floserver_manifest_build PUBLIC floserver_core PRIVATE floserver_third_party floserver_monocypher)
floserver_code_rules(floserver_manifest_build)

add_executable(floserver_manifest main.cpp)
set_target_properties(floserver_manifest PROPERTIES OUTPUT_NAME floserver-manifest)
target_link_libraries(floserver_manifest PRIVATE floserver_manifest_build floserver_core floserver_third_party floserver_monocypher)
if(WIN32)
    target_link_libraries(floserver_manifest PRIVATE bcrypt advapi32)
endif()
floserver_code_rules(floserver_manifest)
